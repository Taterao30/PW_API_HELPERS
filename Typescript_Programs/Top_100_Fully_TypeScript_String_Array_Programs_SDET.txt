15. Check Anagram Without sort()
--------------------------------
function isAnagram(a: string, b: string): boolean {
    if (a.length !== b.length) {
        return false;
    }

    const frequency: Map<string, number> = new Map<string, number>();

    for (const char of a) {
        frequency.set(char, (frequency.get(char) ?? 0) + 1);
    }

    for (const char of b) {
        const count: number | undefined = frequency.get(char);

        if (count === undefined || count === 0) {
            return false;
        }

        frequency.set(char, count - 1);
    }

    return true;
}

console.log(isAnagram("listen", "silent"));

Output:
true

39. Longest Substring Without Repeating Characters
--------------------------------------------------
function longestUnique(str: string): number {
    const seen: Set<string> = new Set<string>();
    let left: number = 0;
    let maxLength: number = 0;

    for (let right: number = 0; right < str.length; right++) {
        while (seen.has(str[right])) {
            seen.delete(str[left]);
            left++;
        }

        seen.add(str[right]);
        maxLength = Math.max(maxLength, right - left + 1);
    }

    return maxLength;
}

console.log(longestUnique("abcabcbb"));

Output:
3

80. Two Sum - Return Indexes
----------------------------
function twoSum(nums: number[], target: number): number[] {
    const map: Map<number, number> = new Map<number, number>();

    for (let i: number = 0; i < nums.length; i++) {
        const required: number = target - nums[i];
        const index: number | undefined = map.get(required);

        if (index !== undefined) {
            return [index, i];
        }

        map.set(nums[i], i);
    }

    return [];
}

const result: number[] = twoSum([2, 7, 11, 15], 9);
console.log(result);

Output:
[0,1]

function flatten(arr: NestedNumberArray): number[] {
    const result: number[] = [];

    for (const item of arr) {
        if (Array.isArray(item)) {
            result.push(...flatten(item));
        } else {

99. Maximum Sum Subarray - Kadane Algorithm
------------------------------------------
function maxSubArray(nums: number[]): number {
    let currentSum: number = nums[0];
    let maxSum: number = nums[0];

    for (let i: number = 1; i < nums.length; i++) {
        currentSum = Math.max(nums[i], currentSum + nums[i]);
        maxSum = Math.max(maxSum, currentSum);
    }

    return maxSum;
}

const result: number = maxSubArray(
    [-2, 1, -3, 4, -1, 2, 1, -5, 4]
);

console.log(result);

Output:
6


100. Longest Consecutive Sequence
---------------------------------
function longestConsecutive(nums: number[]): number {
    const numbers: Set<number> = new Set<number>(nums);
    let longest: number = 0;

    for (const num of numbers) {
        if (!numbers.has(num - 1)) {
            let current: number = num;
            let length: number = 1;

            while (numbers.has(current + 1)) {
                current++;
                length++;
            }

            longest = Math.max(longest, length);
        }
    }

    return longest;
}

const result: number = longestConsecutive(
    [100, 4, 200, 1, 3, 2]
);

console.log(result);

Output:
4
