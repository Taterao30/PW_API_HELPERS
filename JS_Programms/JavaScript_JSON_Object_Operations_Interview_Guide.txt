JAVASCRIPT JSON OBJECT & ARRAY OPERATIONS - SDET INTERVIEW GUIDE
===========================================================
For Playwright / TypeScript / API Automation Interviews
Run examples separately in Node.js or browser console.

IMPORTANT BASICS
----------------
- JSON is a text format; JavaScript objects and arrays are in-memory values.
- JSON.parse(jsonText) converts JSON text to a JS value.
- JSON.stringify(value) converts a JS value to JSON text.
- find() returns the first matching element or undefined.
- filter() returns an array of all matching elements.
- some() checks whether at least one element matches; every() checks all.
- sort() mutates the original array. Use [...arr].sort() to preserve it.
- Comparisons involving strings can be case-sensitive; normalize as needed.
- JSON has no undefined, functions, or comments.

  { id: 102, name: 'Neha', department: 'Dev', salary: 95000, active: true,

  { id: 103, name: 'Ravi', department: 'QA', salary: 80000, active: false,
    skills: ['Selenium', 'Java'], address: { city: 'Pune', pin: 411002 } },
  { id: 104, name: 'Sara', department: 'QA', salary: 95000, active: true,
    skills: ['Playwright', 'TypeScript'], address: { city: 'Bengaluru', pin: 560001 } },
  { id: 105, name: 'Iqbal', department: 'Dev', salary: 60000, active: true,

01. PARSE A JSON RESPONSE
-------------------------
const jsonText = '{"id":101,"name":"Amit","active":true}';
const obj = JSON.parse(jsonText);
console.log(obj.name, obj.active);
// Output: Amit true

02. CONVERT OBJECT TO JSON STRING
---------------------------------
const obj = { id: 101, name: 'Amit' };
console.log(JSON.stringify(obj));
// Output: {"id":101,"name":"Amit"}

03. ACCESS PROPERTY USING DOT / BRACKETS
----------------------------------------
console.log(employees[0].name);
console.log(employees[0]['department']);
// Output: Amit; QA

04. SEARCH EMPLOYEE BY ID (find)
--------------------------------
const found = employees.find(e => e.id === 103);
console.log(found?.name ?? 'Not found');
// Output: Ravi

05. SEARCH BY NAME (case insensitive)
-------------------------------------
const found = employees.find(e => e.name.toLowerCase() === 'neha');

// Output: 102

06. FILTER ALL QA EMPLOYEES
---------------------------
const result = employees.filter(e => e.department === 'QA');

07. FILTER USING MULTIPLE CONDITIONS (AND)
------------------------------------------

// Output: [ 'Amit', 'Sara' ]

08. FILTER USING OR CONDITION

-----------------------

12. CHECK IF ANY MATCH EXISTS (some)
------------------------------------

15. SORT BY SALARY ASCENDING (no mutation)
-----------------------------------------
const sorted = [...employees].sort((a, b) => a.salary - b.salary);
console.log(sorted.map(e => `${e.name}:${e.salary}`));
// Output: [ 'Iqbal:60000', 'Amit:70000', 'Ravi:80000', 'Neha:95000', 'Sara:95000' ]

16. SORT BY SALARY DESCENDING
-----------------------------
const sorted = [...employees].sort((a, b) => b.salary - a.salary);
console.log(sorted.map(e => e.name));
// Output: [ 'Neha', 'Sara', 'Ravi', 'Amit', 'Iqbal' ]

17. SORT BY NAME ALPHABETICALLY
-------------------------------
const sorted = [...employees].sort((a, b) => a.name.localeCompare(b.name));
console.log(sorted.map(e => e.name));
// Output: [ 'Amit', 'Iqbal', 'Neha', 'Ravi', 'Sara' ]

18. SORT BY MULTIPLE FIELDS (department then salary DESC)
---------------------------------------------------------

  a.department.localeCompare(b.department) || b.salary - a.salary
);
console.log(sorted.map(e => `${e.department}:${e.name}`));
// Output: [ 'Dev:Neha', 'Dev:Iqbal', 'QA:Sara', 'QA:Ravi', 'QA:Amit' ]

35. REMOVE DUPLICATES BY ID FROM API RESULTS
--------------------------------------------

36. COMPARE TWO JSON OBJECTS IGNORING KEY ORDER
------------------------------------------------

------------------------------------

41. SEARCH NESTED RECORD AND RETURN PARENT ORDER
------------------------------------------------

    { orderId: 'O1', items: [{ sku: 'A' }] },
    { orderId: 'O2', items: [{ sku: 'B' }, { sku: 'C' }] }
  ]

const order = response.orders.find(o => o.items.some(item => item.sku === 'C'));
console.log(order?.orderId);
// Output: O2

42. SEARCH WITH LOOPS AND BREAK AT FIRST MATCH
----------------------------------------------
const response = { orders: [
  { id: 1, items: [{ sku: 'A' }] },
  { id: 2, items: [{ sku: 'B' }, { sku: 'C' }] }
] };
let found = null;
outer: for (const order of response.orders) {
  for (const item of order.items) {
    if (item.sku === 'C') {
      found = { orderId: order.id, item };
      break outer;
    }
  }
}
console.log(found);
// Output: { orderId: 2, item: { sku: 'C' } }

49. PLAYWRIGHT API: SEARCH RESPONSE DATA
----------------------------------------
const { test, expect } = require('@playwright/test');
test('search users in API response', async ({ request }) => {
  const response = await request.get('https://jsonplaceholder.typicode.com/users');
  expect(response.status()).toBe(200);
  const users = await response.json();
  expect(Array.isArray(users)).toBe(true);
  const user = users.find(u => u.id === 3);
  expect(user).toBeDefined();
  expect(user).toHaveProperty('email');
});
// Expected result: Test passes if service responds with expected data.

50. PLAYWRIGHT API: VALIDATE ALL RECORDS
----------------------------------------
const { test, expect } = require('@playwright/test');
test('every user has required fields', async ({ request }) => {
  const response = await request.get('https://jsonplaceholder.typicode.com/users');
  expect(response.ok()).toBeTruthy();
  const users = await response.json();
  expect(users.length).toBeGreaterThan(0);
  for (const user of users) {
    expect(user.id).toEqual(expect.any(Number));
    expect(user.name).toEqual(expect.any(String));
    expect(user.email).toEqual(expect.any(String));
    expect(user.email).toContain('@');
  }
});
// Expected result: Test passes if all users satisfy the assertions.

BONUS - COMMON INTERVIEW DIFFERENCES
------------------------------------
find       -> first matching object, or undefined
filter     -> array of all matching objects (could be [])
findIndex  -> index of first match, or -1
map        -> new transformed array
forEach    -> iteration side effects; returns undefined
reduce     -> accumulate into number/object/Map/array
some       -> true if any item matches
 every     -> true if every item matches (true for empty array)
includes   -> exact-value presence in array/string
sort       -> changes original array unless copied first
slice      -> non-mutating section of an array
splice     -> mutates array (remove/insert/replace)
flatMap    -> map then flatten one level
Object.keys / values / entries -> object inspection
Object.hasOwn -> own-property existence
?.         -> optional chaining
??         -> fallback only for null or undefined
