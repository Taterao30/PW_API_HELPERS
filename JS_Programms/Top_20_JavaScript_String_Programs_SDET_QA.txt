TOP 20 JAVASCRIPT STRING PROGRAMS FOR QA / SDET (4 YEARS EXPERIENCE)
=================================================================
Each example can be run independently using Node.js.

1. REVERSE A STRING WITHOUT reverse()
-------------------------------------
const str = "Playwright";
let reversed = "";
for (let i = str.length - 1; i >= 0; i--) {
    reversed += str[i];
}
console.log(reversed);
// Output: thgirwyalP

2. CHECK PALINDROME
-------------------
const str = "madam";
const reversed = str.split("").reverse().join("");
console.log(str === reversed ? "Palindrome" : "Not Palindrome");
// Output: Palindrome

3. COUNT VOWELS AND CONSONANTS
-----------------------------
const str = "automation";
let vowels = 0;
let consonants = 0;
for (const char of str.toLowerCase()) {
    if ("aeiou".includes(char)) vowels++;
    else if (/[a-z]/.test(char)) consonants++;
}
console.log("Vowels:", vowels);
console.log("Consonants:", consonants);
// Output: Vowels: 6
//         Consonants: 4

4. FIND DUPLICATE CHARACTERS
---------------------------
const str = "programming";
const count = {};
for (const char of str) {
    count[char] = (count[char] || 0) + 1;
}
for (const char in count) {
    if (count[char] > 1) console.log(char, count[char]);
}
// Output: r 2
//         g 2
//         m 2

5. FIRST NON-REPEATING CHARACTER
-------------------------------
const str = "swiss";
const count = {};
for (const char of str) {
    count[char] = (count[char] || 0) + 1;
}
for (const char of str) {
    if (count[char] === 1) {
        console.log(char);
        break;
    }
}
// Output: w

6. CHARACTER FREQUENCY
----------------------
const str = "testing";
const frequency = {};
for (const char of str) {
    frequency[char] = (frequency[char] || 0) + 1;
}
console.log(frequency);
// Output: { t: 2, e: 1, s: 1, i: 1, n: 1, g: 1 }

7. REMOVE DUPLICATE CHARACTERS
------------------------------
const str = "automation";
let result = "";
for (const char of str) {
    if (!result.includes(char)) result += char;
}
console.log(result);
// Output: automin

8. REVERSE EACH WORD
--------------------
const str = "JavaScript Automation";
const result = str.split(" ")
    .map(word => word.split("").reverse().join(""))
    .join(" ");
console.log(result);
// Output: tpircSavaJ noitamotuA

9. REVERSE WORD ORDER
---------------------
const str = "I love Playwright";
const result = str.split(" ").reverse().join(" ");
console.log(result);
// Output: Playwright love I

10. CHECK ANAGRAM
-----------------
const str1 = "listen";
const str2 = "silent";
const sortString = str => str.split("").sort().join("");
console.log(sortString(str1) === sortString(str2) ? "Anagram" : "Not Anagram");
// Output: Anagram

11. LONGEST WORD IN A SENTENCE
------------------------------
const str = "I am learning JavaScript automation";
const words = str.split(" ");
let longest = "";
for (const word of words) {
    if (word.length > longest.length) longest = word;
}
console.log(longest);
// Output: JavaScript

12. COUNT WORDS (EXTRA SPACES)
-----------------------------
const str = "  Playwright   API  Automation  ";
const words = str.trim() ? str.trim().split(/\s+/) : [];
console.log(words.length);
// Output: 3

13. REMOVE ALL WHITESPACE
-------------------------
const str = "Playwright API Testing";
const result = str.replace(/\s+/g, "");
console.log(result);
// Output: PlaywrightAPITesting

14. COUNT SUBSTRING OCCURRENCES
-------------------------------
const str = "test automation test api test";
const matches = str.match(/test/g);
console.log(matches ? matches.length : 0);
// Output: 3

15. MOST FREQUENT CHARACTER
---------------------------
const str = "success";
const frequency = {};
for (const char of str) {
    frequency[char] = (frequency[char] || 0) + 1;
}
let maxChar = "";
let maxCount = 0;
for (const char in frequency) {
    if (frequency[char] > maxCount) {
        maxChar = char;
        maxCount = frequency[char];
    }
}
console.log(maxChar, maxCount);
// Output: s 3

16. CAPITALIZE FIRST LETTER OF EACH WORD
----------------------------------------
const str = "playwright automation testing";
const result = str.split(" ")
    .map(word => word.charAt(0).toUpperCase() + word.slice(1))
    .join(" ");
console.log(result);
// Output: Playwright Automation Testing

17. LONGEST SUBSTRING WITHOUT REPEATING CHARACTERS
--------------------------------------------------
const str = "abcabcbb";
let start = 0;
let longest = "";
const seen = new Map();
for (let end = 0; end < str.length; end++) {
    const char = str[end];
    if (seen.has(char) && seen.get(char) >= start) {
        start = seen.get(char) + 1;
    }
    seen.set(char, end);
    const current = str.slice(start, end + 1);
    if (current.length > longest.length) longest = current;
}
console.log(longest);
console.log(longest.length);
// Output: abc
//         3

18. EXTRACT NUMBERS FROM STRING
-------------------------------
const str = "Order123Amount450";
const numbers = str.match(/\d+/g) || [];
console.log(numbers);
// Output: [ '123', '450' ]
// QA use: Extract order numbers or amounts from UI messages.

19. BASIC EMAIL FORMAT VALIDATION
---------------------------------
const email = "qa.test@example.com";
const regex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
console.log(regex.test(email) ? "Valid Email" : "Invalid Email");
// Output: Valid Email
// Note: This is a basic format check, not complete RFC validation.

20. COMPARE STRINGS IGNORING CASE AND EXTRA SPACES
-------------------------------------------------
const expected = "Login Successful";
const actual = "  login   successful  ";
function normalizeText(str) {
    return str.trim().replace(/\s+/g, " ").toLowerCase();
}
console.log(normalizeText(expected) === normalizeText(actual)
    ? "Text Matched" : "Text Not Matched");
// Output: Text Matched
// QA use: Compare expected and actual UI messages in Playwright.
