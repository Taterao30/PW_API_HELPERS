	public static void main(String[] args) {

//    	Explanation
//    	1. str.indexOf(c) != str.lastIndexOf(c) — Checks whether the character is repeated.
//    	2. str.indexOf(c) == i — Ensures the character is counted only once.
//    	3. count++ — Counts distinct duplicate characters.
//    	4. count == 2 — Finds the second distinct duplicate character.
//    	5. break — Stops the loop once the result is found.

		String str = "abbcddeffgh";

		int count = 0;

		for (int i = 0; i < str.length(); i++) {

			char c = str.charAt(i);

			if (str.indexOf(c) != str.lastIndexOf(c) && str.indexOf(c) == i) {

				count++;

				if (count == 2) {
					System.out.println("Second duplicate: " + c);
					break;
				}
			}
		}
	}
}

public class ReverseOnlyChar {

	public static void main(String[] args) {

		String str = "suraj10gaikwad20";

		char[] ch = str.toCharArray();

		int left = 0;
		int right = str.length() - 1;

		while (left < right) {

			if (!Character.isLetter(ch[left])) {
				left++;
			}

			else if (!Character.isLetter(ch[right])) {
				right--;
			}

			else {
				char temp = ch[left];
				ch[left] = ch[right];
				ch[right] = temp;

				left++;
				right--;
			}
		}

		System.out.println(new String(ch));
	}
}
//We use left and right pointers to reverse characters from both ends of the string.
//1. left = 0 → Points to the first character.
//2. right = str.length() - 1 → Points to the last character.
//3. We skip digits and swap letters from both ends.
//4. After swapping, left++ moves forward and right-- moves backward.

public class SecondMostFrequentCharInString {

	// c appears 4 times → Second highest

	public static void main(String[] args) {
		
		String input = "aabbbccccddddd";

		HashMap<Character, Integer> hMap = new HashMap<Character, Integer>();

		for (char inputChar : input.toCharArray()) {

			hMap.put(inputChar, hMap.getOrDefault(inputChar, 0) + 1);
		}

		System.out.println(hMap);

		// Find the 2nd most frequent character in the String!!
		int firstMaxFrequency = 0;
		int secondMaxFrequency = 0;

		for (Map.Entry<Character, Integer> entrySet : hMap.entrySet()) {

			int currentFrequency = entrySet.getValue();

			if (currentFrequency > firstMaxFrequency) {

				secondMaxFrequency = firstMaxFrequency;
				firstMaxFrequency = currentFrequency;

			} else if (currentFrequency > secondMaxFrequency && (currentFrequency < firstMaxFrequency)) {

				secondMaxFrequency = currentFrequency;
			}
		}

		System.out.println(secondMaxFrequency);

		for (Map.Entry<Character, Integer> entrySet : hMap.entrySet()) {

			if (entrySet.getValue() == secondMaxFrequency) {

				System.out.println(entrySet.getKey());
			}
		}
	}
}

package October_2026_Programms;

import java.util.Arrays;

public class WordReversal {

	public static void main(String[] args) {

		String input = "Java Test";
		StringBuilder finalResult = new StringBuilder();

		String data[] = input.split(" ");
		System.out.println(Arrays.toString(data));

		for (String word : data) {

			char charArray[] = word.toCharArray();

			int left = 0;
			int right = charArray.length - 1;

			while (left < right) {

				char temp = charArray[left];
				charArray[left] = charArray[right];
				charArray[right] = temp;

				left++;
				right--;
			}

			String result = new String(charArray);

			finalResult.append(result).append(" ");
		}

		System.out.println(finalResult);
	}
}
