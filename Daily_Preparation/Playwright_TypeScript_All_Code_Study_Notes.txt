type BrowserName = 'chrome' | 'edge' | 'firefox' | 'webkit';

async function launchBrowser(browserName: BrowserName) {
    console.log('browser name: ', browserName);

    switch (browserName.trim().toLowerCase()) {
        case 'chrome':
            return await chromium.launch({ headless: false, channel: 'chrome' });
        case 'edge':
            return await chromium.launch({ headless: false, channel: 'msedge' });
        case 'firefox':
            return await firefox.launch({ headless: false });
        case 'webkit':
            return await webkit.launch({ headless: false });
        default:
            console.log('invalid browser....', browserName);
            throw new Error(`invalid browser:, ${browserName}`);
    }
}

//calling this :
let browser: Browser = await launchBrowser('chrome');
let page: Page = await browser.newPage();
await page.goto('https://naveenautomationlabs.com/opencart/index.php?route=account/login');
let title = await page.title();
console.log('page title: ', title);

function getBilling(foodBill: number, tax?: number, vat: number=90): void {

    if (tax && vat) {
        console.log(foodBill + tax + vat);
    }
    else if (tax) {
        console.log(foodBill + tax);
    }
    else {
        console.log(foodBill);
    }

}

getBilling(1000, 20, 5);
getBilling(1000, 500);
getBilling(1000);
getBilling(100, 0, 5);

test('multiple windows handle test', async ({ browser }) => {

    let context = await browser.newContext();
    let page = await context.newPage();
    await page.goto('https://orangehrm.com/contact-sales');

    //array of locators/links:
    let links: Locator[] = [
        page.getByRole('link', { name: 'About Us' }),
        page.getByRole('link', { name: 'Contact Us' }),
        page.getByRole('link', { name: 'Press Releases' }),
        page.getByRole('link', { name: 'Become a Partner' })
    ];

    let childWindowPages: Page[] = []; //0 --> 4
    for (let link of links) {
        let [childWindow] = await Promise.all([
            context.waitForEvent('page'),
            link.click()
        ]);

        await childWindow.waitForLoadState();
        childWindowPages.push(childWindow);
    }

    console.log('total number of pages: ', context.pages().length);//5

    //go to each and every child winow and get the title, close it...

    for (let i = 0; i < childWindowPages.length; i++) {
        let child = childWindowPages[i];
        await child.bringToFront();
        console.log(await child.title());
        await child.waitForTimeout(1000);
        await child.close();
    }

    //back to parent window...only 1 tab is left...

    await page.bringToFront();
    console.log('parent window title: ', await page.title());
    console.log('total number of pages: ', context.pages().length); //1


    await page.pause();

});

import { Browser, BrowserContext, Page, chromium, expect, firefox, test, webkit } from '@playwright/test';

//01: single selection: username is found on the 1st page, on any page, on the last page, user is not found at all
//02: multiple selection: usernames are found on the 1st page, on any page, on the last page, users are not found at all


test('Webtable with pagination with single selection test', async ({ page }) => {
    await page.goto('https://naveenautomationlabs.com/opencart/ui/webtable-pagination.html');

    while (true) {

        let isUserPresent = await page.locator(`td:has-text("cameron_ray")`).first().isVisible();
        if (isUserPresent) {
            //user is found on the 1st page itself..
            await page.locator('table#dataTable tr').filter({ hasText: 'cameron_ray' }).first().getByRole('checkbox').check();
            break;
        }
        else {
            //username is not present on the 1st page, pagination logic:
            //click on the next icon:
            let nextEle = page.getByRole('button', { name: '›' });
            await nextEle.click();
            await page.waitForTimeout(1500);

            let isNextEleDisabled = await nextEle.isDisabled();
            if (isNextEleDisabled) {
                console.log('pagination is over...username is not found...');
                break;
            }
        }


    }
    //await page.pause();

});


test('Webtable with pagination with multiple selection test', async ({ page }) => {
    await page.goto('https://naveenautomationlabs.com/opencart/ui/webtable-pagination.html');

    while (true) {

        let allUserChecks =
            await page.locator('table#dataTable tr').filter({ hasText: 'john_doe' }).getByRole('checkbox').all();

        if (allUserChecks.length > 0) {
            //select all the checkboxes:
            for (let ele of allUserChecks) {
                await ele.check();
            }
        }
        //go the next page, click on next icon:
        let nextEle = page.getByRole('button', { name: '›' });

        let isNextEleDisabled = await nextEle.isDisabled();
        if (isNextEleDisabled) {
            console.log('pagination is over...');
            break;
        }
        await page.waitForTimeout(2000);
        await nextEle.click();
    }
    await page.pause();

});
